% AEGIS box colour palette
% ----------------------------------------------------------

% ----------------------------------------------------------
% Shared box style
% ----------------------------------------------------------
\tcbset{
  aegisbox/.style={
    enhanced,
    breakable,
    sharp corners,
    boxrule=0.8pt,
    left=1.5mm,
    right=1.5mm,
    top=1mm,
    bottom=1mm,
    before skip=8pt,
    after skip=8pt,
    fonttitle=\bfseries,
    coltitle=white
  }
}

\newtcolorbox{todobox}[1][To be completed]{
  aegisbox,
  colback=AEGISYellow,
  colframe=AEGISDarkYellow,
  title=#1
}
